cmake_minimum_required(VERSION 3.24)
project(mika_continuous_experiment LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(mika-segmenter SHARED segmenter.c)
target_compile_definitions(mika-segmenter PUBLIC MIKA_SEGMENTER_SHARED PRIVATE MIKA_SEGMENTER_BUILD)
target_include_directories(mika-segmenter PUBLIC .)
add_executable(mika-segmenter-unit unit.c)
target_link_libraries(mika-segmenter-unit PRIVATE mika-segmenter)
foreach(target mika-segmenter mika-segmenter-unit)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror)
    endif()
endforeach()
add_test(NAME segmenter COMMAND mika-segmenter-unit)
set_tests_properties(segmenter PROPERTIES TIMEOUT 30)
