cmake_minimum_required(VERSION 3.24)
project(mika_model_free LANGUAGES C CXX)
# Separate developer entry point, not an alternate application configuration.
# No FetchContent, microphone, model conversion, or external Python packages.
get_filename_component(MIKA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# A release-version edit must also regenerate an existing model-free build.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MIKA_ROOT}/CMakeLists.txt")
file(STRINGS "${MIKA_ROOT}/CMakeLists.txt" version_line REGEX "^project\\(mika VERSION ")
string(REGEX REPLACE ".*VERSION ([^ ]+).*" "\\1" PROJECT_VERSION "${version_line}")
configure_file("${MIKA_ROOT}/src/mika_version.h.in" generated/mika_version.h @ONLY)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
find_package(CURL 7.85 REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
include("${MIKA_ROOT}/cmake/Core.cmake")
include("${MIKA_ROOT}/cmake/UnitTests.cmake")

foreach(unit IN ITEMS punctuation fullstop)
    if(unit STREQUAL "punctuation")
        add_executable(mika-${unit}-unit "${MIKA_ROOT}/tests/punctuation_unit.c")
        target_link_libraries(mika-${unit}-unit PRIVATE mika_punctuation_guard)
    else()
        add_executable(mika-${unit}-unit "${MIKA_ROOT}/tests/fullstop_unit.cpp")
        target_link_libraries(mika-${unit}-unit PRIVATE mika_fullstop_render)
    endif()
    mika_warnings(mika-${unit}-unit)
    add_test(NAME ${unit}-unit COMMAND mika-${unit}-unit)
endforeach()
add_executable(mika-unit "${MIKA_ROOT}/tests/unit.c")
target_link_libraries(mika-unit PRIVATE mika_model mika_pcm)
mika_warnings(mika-unit)
add_test(NAME unit COMMAND mika-unit "${CMAKE_BINARY_DIR}/test-data/unit")
add_executable(mika-download-test "${MIKA_ROOT}/tests/download.c")
target_link_libraries(mika-download-test PRIVATE mika_model)
mika_warnings(mika-download-test)

add_library(mika_punctuation STATIC "${MIKA_ROOT}/src/punctuation_backend.cpp")
target_link_libraries(mika_punctuation PUBLIC mika_punctuation_guard mika_verify mika_assets ${CMAKE_DL_LIBS})
mika_warnings(mika_punctuation)
if(MSVC)
    target_compile_options(mika_punctuation PRIVATE /EHsc-)
endif()
add_library(mika_cli_stubs STATIC "${MIKA_ROOT}/tests/cli_stubs.c")
target_compile_definitions(mika_cli_stubs PRIVATE mika_model_ensure=mika_fixture_model_ensure)
target_link_libraries(mika_cli_stubs PUBLIC mika_pcm mika_platform)
mika_warnings(mika_cli_stubs)
foreach(cli IN ITEMS fixture model-test)
    add_executable(mika-cli-${cli} "${MIKA_ROOT}/src/main.c")
    target_link_libraries(mika-cli-${cli} PRIVATE mika_cli_stubs mika_punctuation mika_model)
    mika_warnings(mika-cli-${cli})
endforeach()
target_compile_definitions(mika-cli-fixture PRIVATE mika_model_ensure=mika_fixture_model_ensure)
add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${MIKA_ROOT}/tests/cli.py"
    "$<TARGET_FILE:mika-cli-model-test>" "$<TARGET_FILE:mika-cli-fixture>")
add_test(NAME download COMMAND "${Python3_EXECUTABLE}" "${MIKA_ROOT}/tests/download.py"
    "$<TARGET_FILE:mika-download-test>")
foreach(test IN ITEMS metadata package punctuation_evaluation punctuation_fetch spoken_evaluation benchmark_tools speech_assets release_assets release_publish)
    add_test(NAME ${test} COMMAND "${Python3_EXECUTABLE}" "${MIKA_ROOT}/tests/${test}.py")
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_test(NAME installers COMMAND "${Python3_EXECUTABLE}" "${MIKA_ROOT}/tests/installers.py"
        "${MIKA_ROOT}/install.sh")
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES TIMEOUT 90 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
